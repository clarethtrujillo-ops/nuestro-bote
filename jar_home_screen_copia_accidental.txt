class JarHomeScreen extends StatefulWidget {
  const JarHomeScreen({super.key});

  @override
  State<JarHomeScreen> createState() => _JarHomeScreenState();
}

class _JarHomeScreenState extends State<JarHomeScreen> {
  int _selectedIndex = 0;
  Wish? _latestWish;

  void _goBack() {
    final navigator = Navigator.of(context);

    if (navigator.canPop()) {
      navigator.pop();
      return;
    }

    navigator.pushReplacementNamed(
      AppRoutes.connectionSuccess,
    );
  }

  void _selectDestination(int index) {
    if (index == 0) {
      setState(() {
        _selectedIndex = index;
      });
      return;
    }

    ScaffoldMessenger.of(context)
      ..hideCurrentSnackBar()
      ..showSnackBar(
        SnackBar(
          content: Text(
            index == 1
                ? 'La vista de Recuerdos será creada después.'
                : 'La vista de Perfil será creada después.',
          ),
          behavior: SnackBarBehavior.floating,
          backgroundColor: AppColors.elevated,
        ),
      );
  }

  Future<void> _openNewWish() async {
    final createdWish = await Navigator.of(context).pushNamed<Wish>(
      AppRoutes.createWish,
    );

    if (!mounted || createdWish == null) {
      return;
    }

    setState(() {
      _latestWish = createdWish;
    });
  }

  Future<void> _openWishDetails() async {
    final currentWish = _latestWish;

    if (currentWish == null) {
      return;
    }

    final updatedWish = await Navigator.of(context).pushNamed<Wish>(
      AppRoutes.wishDetail,
      arguments: currentWish,
    );

    if (!mounted || updatedWish == null) {
      return;
    }

    setState(() {
      _latestWish = updatedWish;
    });
  }

  @override
  Widget build(BuildContext context) {
    final wish = _latestWish;

    return Scaffold(
      backgroundColor: AppColors.background,
      bottomNavigationBar: JarBottomNavigation(
        selectedIndex: _selectedIndex,
        onDestinationSelected: _selectDestination,
      ),
      body: SafeArea(
        bottom: false,
        child: Center(
          child: ConstrainedBox(
            constraints: const BoxConstraints(
              maxWidth: 430,
            ),
            child: wish == null
                ? _EmptyJarView(
                    onBack: _goBack,
                    onCreateWish: _openNewWish,
                  )
                : _PopulatedJarView(
                    wish: wish,
                    onBack: _goBack,
                    onAddWish: _openNewWish,
                    onOpenWish: _openWishDetails,
                  ),
          ),
        ),
      ),
    );
  }
}

class _PixelHeartPainter extends CustomPainter {
  const _PixelHeartPainter();

  @override
  void paint(Canvas canvas, Size size) {
    final unit = size.width / 45;

    final heartPaint = Paint()
      ..color = AppColors.lavender.withValues(
        alpha: 0.20,
      );

    final darkHeartPaint = Paint()
      ..color = AppColors.lavender.withValues(
        alpha: 0.09,
      );

    final coralPaint = Paint()
      ..color = AppColors.coral.withValues(
        alpha: 0.80,
      );

    final pinkPaint = Paint()
      ..color = AppColors.pink.withValues(
        alpha: 0.74,
      );

    final lavenderPaint = Paint()
      ..color = AppColors.lavender.withValues(
        alpha: 0.82,
      );

    void drawPixel(
      double x,
      double y,
      Paint paint, {
      double scale = 0.84,
    }) {
      final pixelSize = unit * scale;

      canvas.drawRect(
        Rect.fromLTWH(
          x,
          y,
          pixelSize,
          pixelSize,
        ),
        paint,
      );
    }

    final trailPixels = <_PixelPosition>[
      const _PixelPosition(0.2, 6.7),
      const _PixelPosition(3.0, 8.4),
      const _PixelPosition(6.2, 10.3),
      const _PixelPosition(9.4, 9.2),
      const _PixelPosition(12.6, 11.9),
      const _PixelPosition(16.2, 10.7),
      const _PixelPosition(19.7, 13.1),
      const _PixelPosition(23.0, 11.4),
      const _PixelPosition(26.0, 13.7),
    ];

    for (var index = 0; index < trailPixels.length; index++) {
      final pixel = trailPixels[index];

      Paint selectedPaint;

      if (index == 0 ||
          index == 2 ||
          index == 4 ||
          index == 6) {
        selectedPaint = coralPaint;
      } else if (index == 7) {
        selectedPaint = lavenderPaint;
      } else {
        selectedPaint = heartPaint;
      }

      drawPixel(
        pixel.column * unit,
        pixel.row * unit,
        selectedPaint,
      );
    }

    final originX = size.width - (18 * unit);
    final originY = unit * 0.4;

    const heartOutline = <_PixelPosition>[
      _PixelPosition(4, 1),
      _PixelPosition(5, 1),
      _PixelPosition(6, 1),
      _PixelPosition(10, 1),
      _PixelPosition(11, 1),
      _PixelPosition(12, 1),
      _PixelPosition(3, 2),
      _PixelPosition(7, 2),
      _PixelPosition(9, 2),
      _PixelPosition(13, 2),
      _PixelPosition(2, 3),
      _PixelPosition(8, 3),
      _PixelPosition(14, 3),
      _PixelPosition(2, 4),
      _PixelPosition(14, 4),
      _PixelPosition(2, 5),
      _PixelPosition(14, 5),
      _PixelPosition(2, 6),
      _PixelPosition(14, 6),
      _PixelPosition(3, 7),
      _PixelPosition(13, 7),
      _PixelPosition(4, 8),
      _PixelPosition(12, 8),
      _PixelPosition(5, 9),
      _PixelPosition(11, 9),
      _PixelPosition(6, 10),
      _PixelPosition(10, 10),
      _PixelPosition(7, 11),
      _PixelPosition(9, 11),
      _PixelPosition(8, 12),
    ];

    for (final pixel in heartOutline) {
      drawPixel(
        originX + (pixel.column * unit),
        originY + (pixel.row * unit),
        heartPaint,
        scale: 0.94,
      );
    }

    const innerPixels = <_PixelPosition>[
      _PixelPosition(5, 3),
      _PixelPosition(6, 3),
      _PixelPosition(10, 3),
      _PixelPosition(11, 3),
      _PixelPosition(4, 4),
      _PixelPosition(12, 4),
      _PixelPosition(4, 5),
      _PixelPosition(12, 5),
      _PixelPosition(5, 6),
      _PixelPosition(11, 6),
      _PixelPosition(6, 7),
      _PixelPosition(10, 7),
      _PixelPosition(7, 8),
      _PixelPosition(9, 8),
    ];

    for (final pixel in innerPixels) {
      drawPixel(
        originX + (pixel.column * unit),
        originY + (pixel.row * unit),
        darkHeartPaint,
        scale: 0.88,
      );
    }

    const coralHighlights = <_PixelPosition>[
      _PixelPosition(2, 5),
      _PixelPosition(9, 5),
      _PixelPosition(9, 6),
    ];

    for (final pixel in coralHighlights) {
      drawPixel(
        originX + (pixel.column * unit),
        originY + (pixel.row * unit),
        coralPaint,
        scale: 0.94,
      );
    }

    const lavenderHighlights = <_PixelPosition>[
      _PixelPosition(14, 5),
      _PixelPosition(12, 7),
      _PixelPosition(10, 9),
    ];

    for (final pixel in lavenderHighlights) {
      drawPixel(
        originX + (pixel.column * unit),
        originY + (pixel.row * unit),
        lavenderPaint,
        scale: 0.94,
      );
    }

    drawPixel(
      originX + (8 * unit),
      originY + (5 * unit),
      pinkPaint,
      scale: 1.65,
    );
  }

  @override
  bool shouldRepaint(
    covariant _PixelHeartPainter oldDelegate,
  ) {
    return false;
  }
}
